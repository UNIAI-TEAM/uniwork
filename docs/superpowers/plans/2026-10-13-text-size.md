```ts
/**
 * Text size — the accessibility setting in Settings → Preferences.
 *
 * Only the NAME of a level crosses into JavaScript. The multiplier each level
 * means lives in `styles/tokens.css` (`--font-scale` under `[data-text-size]`),
 * so the scale has one source and the boot script below carries no numbers.
 * Kept on this device, like the accent and light/dark.
 */

/** Insertion order is the order of the picker. */
export const TEXT_SIZES = ["default", "large", "xlarge"] as const;

export type TextSize = (typeof TEXT_SIZES)[number];

/** The level `:root` already holds; it needs no attribute. */
export const DEFAULT_TEXT_SIZE: TextSize = "default";

export const TEXT_SIZE_STORAGE_KEY = "uniwork-text-size";

export function isTextSize(value: unknown): value is TextSize {
  return typeof value === "string" && (TEXT_SIZES as readonly string[]).includes(value);
}

export function applyTextSize(size: TextSize, root: HTMLElement): void {
  if (size === DEFAULT_TEXT_SIZE) root.removeAttribute("data-text-size");
  else root.setAttribute("data-text-size", size);
}

export function readStoredTextSize(): TextSize {
  try {
    const stored = window.localStorage.getItem(TEXT_SIZE_STORAGE_KEY);
    return isTextSize(stored) ? stored : DEFAULT_TEXT_SIZE;
  } catch {
    // Private mode / blocked storage: the default size is a fine answer.
    return DEFAULT_TEXT_SIZE;
  }
}

/**
 * Runs before first paint from a blocking inline <script>, like
 * `accentBootScript`: without it the page paints one frame at 14px and then
 * jumps. Only the non-default names are accepted, so a stale or hand-edited
 * value leaves the default in place.
 */
export function textSizeBootScript(): string {
  const levels = TEXT_SIZES.filter((size) => size !== DEFAULT_TEXT_SIZE);
  return `(function(){try{var s=localStorage.getItem(${JSON.stringify(TEXT_SIZE_STORAGE_KEY)});if(${JSON.stringify(levels)}.indexOf(s)<0)return;document.documentElement.setAttribute("data-text-size",s);}catch(_){}})()`;
}
```

```ts
import { beforeEach, describe, expect, it, vi } from "vitest";
import { TEXT_SIZE_STORAGE_KEY } from "../lib/text-size";
import { useTextSize } from "./use-text-size";

describe("useTextSize", () => {

    const { result } = renderHook(() => useTextSize());
    act(() => result.current.setTextSize("xlarge"));
```

```ts
 * Reads and writes the text size. Mirrors `useAccent`: the boot script already
 * applied the stored level before React ran, so this hook starts on the
 * default and syncs on mount rather than reading localStorage during render,
 * which would mismatch the server HTML. It writes the DOM only when the user
 * picks, or when another tab does (the `storage` listener).
 */
export function useTextSize(): { textSize: TextSize; setTextSize: (next: TextSize) => void } {
  const [textSize, setTextSizeState] = useState<TextSize>(DEFAULT_TEXT_SIZE);

  useEffect(() => {
    setTextSizeState(readStoredTextSize());
    const onStorage = (event: StorageEvent) => {
      if (event.key !== TEXT_SIZE_STORAGE_KEY) return;
      const next = readStoredTextSize();
      setTextSizeState(next);
      applyTextSize(next, document.documentElement);
    };
    window.addEventListener("storage", onStorage);
    return () => window.removeEventListener("storage", onStorage);
  }, []);

  const setTextSize = useCallback((next: TextSize) => {
    setTextSizeState(next);
    applyTextSize(next, document.documentElement);
    try {
      window.localStorage.setItem(TEXT_SIZE_STORAGE_KEY, next);
    } catch {
      // Blocked storage: the size still applies for this session.
    }
  }, []);

  return { textSize, setTextSize };
}
```

```tsx
"use client";

import { useId } from "react";
import { useTranslation } from "react-i18next";
import { useTextSize } from "@uniwork/ui/hooks/use-text-size";
import { TEXT_SIZES, type TextSize } from "@uniwork/ui/lib/text-size";
import { cn } from "@uniwork/ui/lib/utils";
import { SettingsSection } from "./settings-layout";
import { FOCUS_FROM_RADIO } from "./themes-panel";

const LABEL_KEYS = {
  default: "textSizeDefault",
  large: "textSizeLarge",
  xlarge: "textSizeXLarge",
} as const satisfies Record<TextSize, string>;

/**
 * Text size, the accessibility setting beside the two theming axes: every
 * reading step of the type scale grows, layout and icons stay. Native radios
 * behind `sr-only`, as in `ThemesPanel`, so arrow keys and the checked state
 * come from the platform. Each sample carries its own `data-text-size`, so it
 * shows its level's size whatever the page is currently set to.
 */
export function TextSizePanel() {
  const { t } = useTranslation(undefined, { keyPrefix: "settings.preferences" });
  const { textSize, setTextSize } = useTextSize();
  const headingId = useId();

  return (
    <SettingsSection title={<span id={headingId}>{t("textSize")}</span>} description={t("textSizeHint")}>
      <fieldset aria-labelledby={headingId} className="flex flex-wrap gap-2">
        {TEXT_SIZES.map((size) => (
          <label
            key={size}
            className={cn(
              "group flex min-h-9 cursor-pointer items-center gap-2.5 rounded-lg border border-border",
              "bg-surface py-1.5 pr-4 pl-2.5 transition-colors pointer-coarse:min-h-11",
              "hover:border-input",
              FOCUS_FROM_RADIO,
              "has-checked:border-brand has-checked:bg-surface-selected",
            )}
          >
            <input
              type="radio"
              name="uw-text-size"
              value={size}
              checked={textSize === size}
              onChange={() => setTextSize(size)}
              className="sr-only"
            />
            <span data-text-size={size} className="min-w-9 text-center text-title-sm font-semibold" aria-hidden>
              {t("textSizeSample")}
            </span>
            <span className="text-body font-medium">{t(LABEL_KEYS[size])}</span>
          </label>
        ))}
      </fieldset>
    </SettingsSection>
  );
}
```

```ts
import { expect, test, type Page } from "@playwright/test";
import { onboardToWorkspace, workspaceSeed } from "./onboard";

// Spec 2026-10-13-text-size-design.md: the text size picked in Settings →
// Preferences scales every reading step, survives a reload without a flash or
// a hydration error, and leaves no main screen scrolling sideways at the
// largest level.
//
// Requires `make dev`.
const w = workspaceSeed("textsize", "owner", Date.now());

function bodyFontSize(page: Page): Promise<string> {
  return page.evaluate(() => getComputedStyle(document.body).fontSize);
}

test("the largest text size scales the app, survives a reload and keeps screens in bounds", async ({ page }) => {
  const hydrationErrors: string[] = [];
  page.on("console", (msg) => {
    if (msg.type() === "error" && /hydrat/i.test(msg.text())) hydrationErrors.push(msg.text());
  });

  await onboardToWorkspace(page, w);
  const base = `/${w.orgSlug}/${w.wsSlug}`;

  await page.goto(`${base}/settings?tab=preferences`);
  const picker = page.getByRole("group", { name: "Cỡ chữ" });
  await expect(picker).toBeVisible({ timeout: 15_000 });
  expect(await bodyFontSize(page)).toBe("14px");

  // The radio is sr-only; the label is what a person clicks.
  await picker.getByText("Rất lớn", { exact: true }).click();
  await expect(page.locator("html")).toHaveAttribute("data-text-size", "xlarge");
  expect(await bodyFontSize(page)).toBe("18px");

  // Stylesheets that read var(--text-*) — the editor — follow too, not only utilities.
  const viaVariable = await page.evaluate(() => {
    const probe = document.createElement("div");
    probe.style.fontSize = "var(--text-body)";
    document.body.append(probe);
    const size = getComputedStyle(probe).fontSize;
    probe.remove();
    return size;
  });
  expect(viaVariable).toBe("18px");

  await page.reload();
  // The hook never writes the attribute on mount, so only the boot script can have set it.
  await expect(page.locator("html")).toHaveAttribute("data-text-size", "xlarge");
  expect(await bodyFontSize(page)).toBe("18px");
  await expect(page.getByRole("radio", { name: "Rất lớn" })).toBeChecked();

  for (const path of ["", "/tasks", "/chat", "/documents", "/settings?tab=preferences"]) {
    await page.goto(`${base}${path}`);
    await page.waitForLoadState("load");
    await expect(page.locator("html")).toHaveAttribute("data-text-size", "xlarge");
    const overflow = await page.evaluate(() => document.documentElement.scrollWidth - window.innerWidth);
    expect(overflow, `${path || "/"} scrolls sideways at the largest text size`).toBeLessThanOrEqual(0);
  }

  expect(hydrationErrors).toEqual([]);
});
```
